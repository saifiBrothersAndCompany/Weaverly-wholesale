    /* ---------------- 3. GPS GEOFENCE (1.5 KM) ---------------- */
    function calcDistanceKm(lat1, lon1, lat2, lon2) {
      var R = 6371;
      var dLat = ((lat2 - lat1) * Math.PI) / 180;
      var dLon = ((lon2 - lon1) * Math.PI) / 180;
      var a =
        Math.sin(dLat / 2) * Math.sin(dLat / 2) +
        Math.cos((lat1 * Math.PI) / 180) *
          Math.cos((lat2 * Math.PI) / 180) *
          Math.sin(dLon / 2) *
          Math.sin(dLon / 2);
      var c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
      return R * c;
    }

    function renderProducts() {
      var q = (document.getElementById("inSearch").value || "").toLowerCase().trim();
      var grid = document.getElementById("uiProdGrid");
      var html = "";

      var filtered = STORE_DATA.products.filter(function (p) {
        var matchCat = curCat === "All" || p.category === curCat;
        var matchSearch = !q || p.name.toLowerCase().indexOf(q) !== -1 || p.category.toLowerCase().indexOf(q) !== -1;
        return matchCat && matchSearch;
      });

      if (filtered.length === 0) {
        grid.innerHTML = '<div style="grid-column:1/-1; text-align:center; padding:32px; color:#6b7280; font-weight:700;">No wholesale items match your search.</div>';
        return;
      }

      filtered.forEach(function (p) {
        var qty = cart[p.id] || 0;
        var saveAmt = p.mrp > p.price ? p.mrp - p.price : 0;
        var isOos = p.inStock === false;

        var imgContent = p.image
          ? '<img src="' + p.image + '" alt="' + p.name + '" loading="lazy" />'
          : (p.emoji || "🛍️");

        var actionHtml = "";
        if (isOos) {
          actionHtml = '<span style="font-size:11px; font-weight:800; color:#dc2626;">Sold Out</span>';
        } else if (qty === 0) {
          actionHtml = '<button type="button" class="btn-add" onclick="changeQty(' + p.id + ', 1)">ADD</button>';
        } else {
          actionHtml =
            '<div class="qty-box">' +
            '<button type="button" class="qty-btn" onclick="changeQty(' + p.id + ', -1)">−</button>' +
            '<span class="qty-num">' + qty + "</span>" +
            '<button type="button" class="qty-btn" onclick="changeQty(' + p.id + ', 1)">+</button>' +
            "</div>";
        }

        html +=
          '<div class="p-card ' + (isOos ? "oos" : "") + '">' +
          (saveAmt > 0 ? '<div class="p-off">SAVE ₹' + saveAmt + "</div>" : "") +
          '<div class="p-slot-tag">🌙 6-9 PM</div>' +
          '<div>' +
          '<div class="p-img">' + imgContent + "</div>" +
          '<span class="p-unit">' + p.unit + "</span>" +
          '<div class="p-name">' + p.name + "</div>" +
          "</div>" +
          '<div class="p-foot">' +
          '<div class="p-price-wrap">' +
          '<span class="p-price">₹' + p.price + "</span>" +
          (p.mrp > p.price ? '<span class="p-mrp">MRP ₹' + p.mrp + "</span>" : "") +
          "</div>" +
          actionHtml +
          "</div>" +
          "</div>";
      });

      grid.innerHTML = html;
    }

    function changeQty(prodId, delta) {
      var cur = cart[prodId] || 0;
      var next = cur + delta;
      if (next <= 0) {
        delete cart[prodId];
      } else {
        cart[prodId] = next;
      }
      localStorage.setItem(KEY_CART, JSON.stringify(cart));
      renderProducts();
      updateCartUI();
      if (document.getElementById("modalCheckout").classList.contains("open")) {
        renderCheckoutModal();
      }
    }

    function getCartTotals() {
      var itemsCount = 0;
      var subTotal = 0;
      var mrpTotal = 0;

      STORE_DATA.products.forEach(function (p) {
        var q = cart[p.id] || 0;
        if (q > 0) {
          itemsCount += q;
          subTotal += p.price * q;
          mrpTotal += (p.mrp || p.price) * q;
        }
      });

      return {
        itemsCount: itemsCount,
        subTotal: subTotal,
        mrpTotal: mrpTotal,
        savings: Math.max(0, mrpTotal - subTotal)
      };
    }

    function updateCartUI() {
      var t = getCartTotals();
      var minOrder = STORE_DATA.minOrderValue || 499;

      // MOV Progress Bar
      var pct = Math.min(100, Math.round((t.subTotal / minOrder) * 100));
      var movBar = document.getElementById("uiMovBar");
      movBar.style.width = pct + "%";
      document.getElementById("uiMovAmt").textContent = "₹" + t.subTotal + " / ₹" + minOrder;

      if (t.subTotal >= minOrder) {
        movBar.classList.add("ok");
        document.getElementById("uiMovTitle").innerHTML = "✅ <strong>Unlocked Free Evening Delivery (6:00 PM – 9:00 PM)!</strong>";
      } else {
        movBar.classList.remove("ok");
        var diff = minOrder - t.subTotal;
        document.getElementById("uiMovTitle").textContent = "Add ₹" + diff + " more for Evening Wholesale Delivery";
      }

      // Bottom Floating Bar
      document.getElementById("uiCartCount").textContent = t.itemsCount + " ITEMS • 6 PM – 9 PM DROP";
      document.getElementById("uiCartTotal").textContent = "₹" + t.subTotal;

      var ticker = document.getElementById("uiSavingsTicker");
      var cartBar = document.getElementById("uiBottomCartBar");

      if (t.savings > 0) {
        ticker.classList.remove("hidden");
        document.getElementById("uiBarSavedAmt").textContent = "₹" + t.savings;
      } else {
        ticker.classList.add("hidden");
      }

      if (t.subTotal >= minOrder) {
        cartBar.classList.remove("locked-bar");
      } else {
        cartBar.classList.add("locked-bar");
      }
    }

    function triggerDownload(filename, content, mime) {
      var blob = new Blob([content], { type: mime });
      var a = document.createElement("a");
      a.href = URL.createObjectURL(blob);
      a.download = filename;
      document.body.appendChild(a);
      a.click();
      document.body.removeChild(a);
    }
